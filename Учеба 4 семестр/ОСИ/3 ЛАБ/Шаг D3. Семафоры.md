Прежде чем писать код — подумай какие семафоры нужны в схеме с управляющим.

Схема работы такая:

- философ хочет есть — он **сообщает управляющему** и ждёт разрешения
- управляющий смотрит на вилки — если обе свободны, даёт разрешение философу
- философ поел — сообщает управляющему что закончил
- управляющий освобождает вилки и проверяет соседей

Значит нужно организовать два вида общения:

**Философ → управляющий** (разбуди меня, я хочу есть / я закончил): тебе нужен один семафор на котором управляющий спит и ждёт пока кто-то из философов его разбудит

**Управляющий → философ** (ты можешь есть): тебе нужен отдельный семафор для каждого философа — управляющий делает post на семафор конкретного философа, тот просыпается и начинает есть

Итого минимум: 1 + 5 семафоров.

---

Также подумай — как управляющий узнает **кто** именно хочет есть и **что именно** произошло (начал хотеть есть или закончил)? Тебе нужно где-то хранить эту информацию — продумай это отдельно от семафоров.

**control_sema** — на нём спит управляющий поток и ждёт сигналов от философов. Философ делает `sem_post(&control_sema)` чтобы разбудить управляющего и сказать "эй, у меня что-то изменилось".

**control_phil.phil_sema** — это защитный семафор с начальным значением 1, работает как мьютекс. Он защищает `was_do` от одновременного доступа нескольких потоков.

Схема обращения к `was_do` выглядит так:

```
sem_wait(&control_phil.phil_sema);  // захватили мьютекс
phil_array[idx].was_do = EAT;       // меняем данные
sem_post(&control_phil.phil_sema);  // отпустили мьютекс
sem_post(&control_sema);            // будим управляющего
```

То есть сначала защищаем данные, потом отдельно будим управляющего. Это два независимых действия.

#include <stdio.h>
#include <time.h>
#include <stdlib.h>
#include <locale.h>
#include <string.h>
#include <pthread.h>
#include <semaphore.h>
struct timespec start_time;

#define CONTROL 1
#define UN_CONTROL 0
#define EAT 1
#define SLEEP 0

#define FREE 1
#define BUSY 0 

#define COUNT_PHIL 5

#define STOP_PROGRAMM 0

static volatile int global_count_fork[5] = {FREE, FREE, FREE, FREE, FREE};

typedef struct
{
    bool was_control;
    bool was_do;
    int phil_fork_count;
    sem_t phil_sema; // D3

}PHIL_struct;

PHIL_struct phil_array[5];
PHIL_struct control_phil; 
sem_t control_sema; // D3

void clear_phil_struct() // D1
{
    control_phil.was_control = CONTROL;
    for (int i = 0; i<COUNT_PHIL; i++)
    {
        phil_array[i].phil_fork_count = 0;
        phil_array[i].was_do = SLEEP;
        phil_array[i].was_control = UN_CONTROL;
    }
}

#define TOTAL 0
#define PHIL 1
int array_for_total_and_phil[2] = {0}; // 0 idx - TOTAL 1 idx - PHIL

void input_to_array_for_total_and_phil(int total, int phil) // D1
{
   array_for_total_and_phil[TOTAL] = total;
   array_for_total_and_phil[PHIL] = phil;
}

void close_all (pthread_t* threads_phil, pthread_t* threads_control) // D2
{
    pthread_join(*threads_control, NULL);
    for (int i = 0; i < COUNT_PHIL; i++) // wait full stop active threads
    {
        pthread_join(threads_phil[i], NULL);
        sem_destroy(&phil_array[i].phil_sema); // D3
    }
    sem_destroy(&control_phil.phil_sema);
    free(threads_phil);
    free(threads_control);
    sem_destroy(&control_sema);
    
}

void* worker_control(void* param)// D2
{
    return 0; 
}

void* worker_phil(void* param) // D2
{
    return 0; 
}

int main(int argc, char* argv[]) // D1
{
    setlocale(LC_ALL, "Russian"); // D1
    clear_phil_struct();
    if (argc != 3)
    {
        printf("Usage: phil TOTAL PHIL\n");
        return 1;
    }
    int total = atoi(argv[1]); // D1
    int phil  = atoi(argv[2]);
    input_to_array_for_total_and_phil(total, phil); // D1
    clock_gettime(CLOCK_MONOTONIC, &start_time); // D1
    pthread_t* threads_phil = (pthread_t*)malloc(COUNT_PHIL * sizeof(pthread_t)); // D2
    if (threads_phil == NULL)
    {
        printf("Не удалось выделить память под потоки\n");
        free(threads_phil);
        return 1;
    }

    pthread_t* threads_control = (pthread_t*)malloc(1 * sizeof(pthread_t)); // D2
    if (threads_control == NULL)
    {
        printf("Не удалось выделить память под потоки\n");
        free(threads_control);
        free(threads_phil);
        return 1;
    }

    for (int i = 0; i<COUNT_PHIL; i++) // D3
    {
        sem_init(&phil_array[i].phil_sema, 0, 0);
    }
    sem_init(&control_phil.phil_sema, 0, 0);
    sem_init(&control_sema, 0, 1); // like mutex

    pthread_create(threads_control, NULL, worker_control, NULL); // D2
    for (int i = 0; i < COUNT_PHIL; i++)
    {
        pthread_create(&threads_phil[i], NULL, worker_phil, (void*)(size_t)i);
       
    }
    close_all(threads_phil, threads_control);
    return 0; 
}
