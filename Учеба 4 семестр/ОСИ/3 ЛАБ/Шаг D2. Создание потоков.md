Тебе нужно создать 6 потоков — 5 философов и 1 управляющий. Все они создаются разом в main, сразу после clock_gettime.

Вот что нужно сделать:

- добавь глобальный массив описателей потоков — для 5 философов и отдельно один для управляющего
- напиши две заглушки поточных функций — одну для философа, одну для управляющего. Пока они просто сразу возвращают NULL, логику добавим позже
- в main создай все 6 потоков через `pthread_create`
- в main дождись завершения всех 6 потоков через `pthread_join`
- не забудь подключить `pthread.h`

#define UN_CONTROL 0

#define COUNT_PHIL 5

typedef struct
{
    bool was_control;
    bool was_do;
    int phil_fork_count;
}PHIL_struct;

PHIL_struct phil_array[5];
PHIL_struct control_phil; 

void clear_phil_struct() // D1
{
    control_phil.was_control = CONTROL;
    for (int i = 0; i<COUNT_PHIL; i++)
    {
        phil_array[i].phil_fork_count = 0;
        phil_array[i].was_do = SLEEP;
        phil_array[i].was_control = UN_CONTROL;
    }
}

#define TOTAL 0
#define PHIL 1
int array_for_total_and_phil[2] = {0}; // 0 idx - TOTAL 1 idx - PHIL

void input_to_array_for_total_and_phil(int total, int phil) // D1
{
   array_for_total_and_phil[TOTAL] = total;
   array_for_total_and_phil[PHIL] = phil;
}

void close_all (pthread_t* threads_phil, pthread_t* threads_control) // D2
{
    pthread_join(*threads_control, NULL);
    for (int i = 0; i < COUNT_PHIL; i++) // wait full stop active threads
    {
        pthread_join(threads_phil[i], NULL);
    }
    free(threads_phil);
    free(threads_control);
}

    return 0;

void* worker_phil(void* param) // D2

int main(int argc, char* argv[]) // D1
{
    setlocale(LC_ALL, "Russian"); // D1
    clear_phil_struct();
    if (argc != 3)
    {
        printf("Usage: phil TOTAL PHIL\n");
        return 1;
    }
    int total = atoi(argv[1]);
    int phil  = atoi(argv[2]);
    input_to_array_for_total_and_phil(total, phil);
    clock_gettime(CLOCK_MONOTONIC, &start_time);
    pthread_t* threads_phil = (pthread_t*)malloc(COUNT_PHIL * sizeof(pthread_t)); // D2
    if (threads_phil == NULL)
    {
        printf("Не удалось выделить память под потоки\n");
        free(threads_phil);
        return 1;
    }

    pthread_t* threads_control = (pthread_t*)malloc(1 * sizeof(pthread_t));
    if (threads_control == NULL)
    {
        printf("Не удалось выделить память под потоки\n");
        free(threads_control);
        return 1;
    }
    pthread_create(threads_control, NULL, worker_control, NULL);
    for (int i = 0; i < COUNT_PHIL; i++)
    {
        pthread_create(&threads_phil[i], NULL, worker_phil, (void*)(size_t)i);
       
    }
    close_all(threads_phil, threads_control);
