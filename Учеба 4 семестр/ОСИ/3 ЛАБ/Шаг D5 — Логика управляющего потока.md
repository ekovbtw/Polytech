Управляющий работает так:

1. спит на `control_sema` — ждёт пока кто-то из философов его разбудит
2. просыпается — сканирует всех философов и смотрит кто поставил `was_do = EAT`
3. для каждого кто хочет есть — проверяет свободны ли обе его вилки
4. если свободны — занимает вилки и делает `sem_post` на семафор этого философа
5. повторяет пока не истекло TOTAL

Как определить какие вилки у философа с индексом idx:

- левая вилка — это `global_count_fork[idx]`
- правая вилка — это `global_count_fork[(idx + 1) % COUNT_PHIL]`

Цикл управляющего крутится пока не истекло TOTAL — так же как у философов.

**Важный момент:**

Когда философ заканчивает есть и ставит `was_do = SLEEP` — управляющий должен **освободить его вилки**. Значит управляющему нужно различать два события:

- философ хочет есть — занять вилки и разбудить
- философ закончил есть — освободить вилки

void* worker_control(void* param)
{
    struct timespec now;
    clock_gettime(CLOCK_MONOTONIC, &now);
    long elapsed = (now.tv_sec - start_time.tv_sec) * 1000 +
                   (now.tv_nsec - start_time.tv_nsec) / 1000000;
    control_phil.was_do = SLEEP;
    while (elapsed < array_for_total_and_phil[TOTAL])
    {
        sem_wait(&control_sema);
        clock_gettime(CLOCK_MONOTONIC, &now);
        elapsed = (now.tv_sec - start_time.tv_sec) * 1000 +
                  (now.tv_nsec - start_time.tv_nsec) / 1000000;
        int count_eat = 0;
        int eat_ar[COUNT_PHIL] = {0};
        int idx = 0;
        for (int i = 0; i < COUNT_PHIL; i++)
        {
            sem_wait(&control_phil.phil_sema);
            if (phil_array[i].was_do == EAT)
            {
                eat_ar[idx] = i;
                idx++;
                count_eat++;
            }
            else if (phil_array[i].was_do == SLEEP)
            {
                global_count_fork[i] = FREE;
                global_count_fork[(i + 1) % COUNT_PHIL] = FREE;
            }
            sem_post(&control_phil.phil_sema);
        }
        for (int i = 0; i < count_eat; i++)
        {
            int index = eat_ar[i];
            bool check = false;
            if (global_count_fork[index] == FREE)
            {
                if (index == 1)
                {
                    check = true;
                    if (global_count_fork[COUNT_PHIL-1] != FREE)
                    {
                        check = false;
                    }
                }
                else
                {
                    check = true;
                    if (global_count_fork[(index - 1 + COUNT_PHIL) % COUNT_PHIL] != FREE)
                    {
                        check = false;
                    }
                }
                if (check == true)
                {
                    if (index == 1)
                    {
                        global_count_fork[index] = BUSY;
                        global_count_fork[(index + 1) % COUNT_PHIL] = BUSY;
                    }
                    else
                    {
                        global_count_fork[index] = BUSY;
                        global_count_fork[(index - 1 + COUNT_PHIL) % COUNT_PHIL] = BUSY;
                    }
                    sem_post(&phil_array[index].phil_sema);
                }
            }
        }
    }
    return 0;
}
