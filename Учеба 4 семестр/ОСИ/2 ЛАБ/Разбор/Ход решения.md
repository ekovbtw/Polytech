Начнем с первой части tcpclient.cpp на Windows. По сути задания на вход подается файл с форматом: 
```
dd.mm.yyyy dd.mm.yyyy hh:mm:ss Message  
где  
dd.mm.yyyy - дата, например 11.09.2001  
hh:mm:ss - время, например: 23:14:51  
Message - текст неограниченной длины, до конца строки.  
Части записи отделяются друг от друга пробелом (одним).
```

```
#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>

// 0 - ошибка в строке 
// 1 - строка корректна

FILE* file_open() // открываем файл
{
	FILE* file = fopen("file1.txt", "r");
	if (file == NULL)
	{
		printf("Error opening file!\n");
		exit(1);
	}
	return file;
}

int check_date(int day, int month, int year) // проверка корректности дат
{
	if (day < 1 || day > 31 || month < 1 || month > 12 || year < 0)
	{
		return 0;
	}
	if ((day == 31 && month == 4) || (day == 31 && month == 6) || (day == 31 && month == 9) || (day == 31 && month == 11))
	{
		return 0;
	}
	if (month == 2 && day > 29)
	{
		return 0;
	}
	return 1;
}

int check_time(int hour, int minute, int second) // проверка корректности времени
{
	if (hour < 0 || hour > 23 || minute < 0 || minute > 59 || second < 0 || second > 59)
	{
		return 0;
	}
	return 1;
}

int parse_file(char* buffer) // сам парсинг
{
	int day1, month1, year1;
	int day2, month2, year2;
	int hour, minute, second;
	char message[256];
	
	int point; // указатель для message (чтобы отслеживать на сколько переместились)
	if (sscanf(buffer, "%d.%d.%d %d.%d.%d %d:%d:%d %n", &day1, &month1, &year1, &day2, &month2, &year2, &hour, &minute, &second, &point) != 9) // сразу проверка на количество полученных значений 
	{
		return 0;
	}
	
	int status1 = check_date(day1, month1, year1); // проверяем даты
	int status2 = check_date(day2, month2, year2);
	if (status1 == 0 || status2 == 0)
	{
		return 0;
	}
	status1 = check_time(hour, minute, second); // проверяем время
	if (status1 == 0)
	{
		return 0;
	}
	/*printf("Date 1: %d.%d.%d\n", day1, month1, year1);
	printf("Date 2: %d.%d.%d\n", day2, month2, year2);
	printf("Time: %d:%d:%d\n", hour, minute, second);*/
	sscanf(buffer + point, "%[^\n]", message); // все символы кроме \n	(парсим message)
	printf("Message: %s\n", message);
	return 1;
}

int main()
{
	FILE* file = file_open(); 
	int counter = 0;
	char buffer[256]; // массив для строки
	while (fgets(buffer, sizeof(buffer), file) != NULL) // пока считывается строка 
	{
		if (strlen(buffer) == 1) // проверка на длину равную \n
		{
			continue;
		}
		int status = parse_file(buffer); // получаем успех/неудача
		if (status == 0) // проверка status
		{
			continue;
		}
		else
		{
			counter++;
		}
	}
	printf("%d\n", counter);
	fclose(file);
	return 0;
}
```

## ШАГ 2
Дальше сборка протокольного сообщения из распаршенных полей в набор байтов. 
Структура сообщения должна выглядеть так: 
- 4 байта — номер сообщения, 32-битное число в сетевом порядке байт, индексация от 0  
- 1 байт — день первой даты  
- 1 байт — месяц первой даты  
- 2 байта — год первой даты, unsigned short в сетевом порядке байт  
- 1 байт — день второй даты  
- 1 байт — месяц второй даты  
- 2 байта — год второй даты, тоже в сетевом порядке  
- 1 байт — час  
- 1 байт — минута  
- 1 байт — секунда  
- N байт — сам текст сообщения  
- 1 байт — нулевой байт-маркер конца сообщения

